Configurações do projeto Supabase para que todas as estações de trabalho sincronizem as ações, observações e desfechos simultaneamente:

Diagnóstico de Conexão

📋 Ver / Copiar Script SQL das Tabelas (para rodar no SQL Editor do Supabase) 
create table if not exists casos_ativos (
  cd_usuario text primary key,
  dados jsonb not null,
  updated_at timestamptz default now()
);

create table if not exists cofre_casos (
  cd_usuario text primary key,
  dados jsonb not null,
  updated_at timestamptz default now()
);

create table if not exists cofre_logs (
  cd_usuario text primary key,
  itens jsonb default '[]'::jsonb,
  updated_at timestamptz default now()
);

alter publication supabase_realtime add table casos_ativos;
alter publication supabase_realtime add table cofre_casos;
alter publication supabase_realtime add table cofre_logs;

alter table casos_ativos enable row level security;
alter table cofre_casos enable row level security;
alter table cofre_logs enable row level security;

create policy "Acesso livre casos_ativos" on casos_ativos for all using (true) with check (true);
create policy "Acesso livre cofre_casos" on cofre_casos for all using (true) with check (true);
create policy "Acesso livre cofre_logs" on cofre_logs for all using (true) with check (true);
